  background: #f4f4f4;
  padding: 15px;
  border-radius: 8px;
  width: 220px;
  text-align: center;

.hover-link {
  text-decoration: none;
  color: #2c3e50;
  font-weight: bold;
}
.hover-link:hover {
  color: #ffffff;
  background: #3498db;
  padding: 5px 10px;
  border-radius: 4px;
}

/* :focus - highlights element when selected */
.focus-btn {
  padding: 8px 14px;
  border: 2px solid #3498db;
  border-radius: 4px;
  background: white;
  cursor: pointer;
}
.focus-btn:focus {
  outline: none;
  border-color: #e74c3c;
  background: #fdecea;
}

/* :active - state while button is pressed */
.active-btn {
  padding: 8px 14px;
  border: none;
  border-radius: 4px;
  background: #2ecc71;
  color: white;
  cursor: pointer;
}

  background: #27ae60;
  transform: scale(0.95);

}

Q4. Build a CSS Selectors Practice Page
====================================================================

--- q4.html ---
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>CSS Selectors Practice</title>
  <link rel="stylesheet" href="q4.css">
</head>
<body>
  <header>
    <h1>CSS Selectors Practice Page</h1>
  </header>
  <main>
    <p>This paragraph is styled using an <strong>element selector</strong>.</p>
    <p class="highlight">This paragraph uses a <strong>class selector</strong>.</p>
    <p id="unique">This paragraph uses an <strong>ID selector</strong>.</p>

    <a href="https://example.com" target="_blank">External Link (attribute selector)</a>

    <div class="combinator-demo">
      <p>Direct child paragraph (used with &gt; combinator)</p>
      <span>Adjacent sibling span (used with + combinator)</span>
      <p>General sibling paragraph (used with ~ combinator)</p>
      <section>
        <p>Descendant paragraph inside section</p>
      </section>
    </div>

    <ul>
      <li>List item one</li>
      <li>List item two</li>
      <li>List item three</li>
    </ul>
  </main>
  <footer>
    <p>Assignment 8 - Question 4</p>
  </footer>
</body>
</html>

--- q4.css ---
* { box-sizing: border-box; margin: 0; padding: 0; }
body { font-family: Arial, sans-serif; padding: 20px; }
header, footer { text-align: center; padding: 15px; background: #ecf0f1; }
main { padding: 20px; max-width: 700px; margin: 0 auto; }

/* Element selector - targets all <p> tags */
p {
  margin-bottom: 10px;
  line-height: 1.5;
}

/* Class selector - targets elements with class="highlight" */
.highlight {
  background: #fff3cd;
  padding: 5px;
}

/* ID selector - targets the single element with id="unique" */
#unique {
  color: #e74c3c;
  font-weight: bold;
}

/* Attribute selector - targets links opening in a new tab */
a[target="_blank"] {
  color: #2980b9;
  text-decoration: underline;
}

/* Child combinator (>) - direct child <p> of .combinator-demo */
.combinator-demo > p {
  color: #27ae60;
}

/* Adjacent sibling combinator (+) - span immediately after a p */
.combinator-demo p + span {
  color: #8e44ad;
  font-style: italic;
}

/* General sibling combinator (~) - any p sibling after the span */
.combinator-demo span ~ p {
  color: #d35400;
}

/* Descendant combinator (space) - any p inside section, at any depth */
.combinator-demo section p {
  font-weight: bold;
  background: #f4f4f4;
  padding: 4px;
}

ul {
  margin-top: 15px;
  padding-left: 20px;
}

body { font-family: Arial, sans-serif; }
header, footer { text-align: center; padding: 15px; }

.animation-stage {
  display: flex;
  flex-direction: column;
  justify-content: center;
  align-items: center;
  gap: 40px;
  min-height: 70vh;
}

.moving-box {
  width: 60px;
  height: 60px;
  background: #3498db;
  animation: moveBox 2s ease-in-out infinite alternate;
}
@keyframes moveBox {
  from { transform: translateX(-100px); }
  to { transform: translateX(100px); }
}

/* Bouncing ball */
.bouncing-ball {
  width: 50px;
  height: 50px;
  border-radius: 50%;
  background: #e74c3c;
  animation: bounce 1s ease-in-out infinite;
  animation-delay: 0.3s;
}
@keyframes bounce {
  0%, 100% { transform: translateY(0); }
  50% { transform: translateY(-40px); }
}

/* Color-changing text */
.color-text {
  font-size: 24px;
  font-weight: bold;
  animation: colorChange 3s ease-in-out infinite;
  animation-iteration-count: infinite;
  animation-direction: alternate;
}
@keyframes colorChange {
  0%   { color: #e74c3c; }
  33%  { color: #2ecc71; }
  66%  { color: #3498db; }
  100% { color: #9b59b6; }
}
