-------------------- QUESTION 1 (q1.html) --------------------------
Embed a YouTube video using percentage width/height.

<html>
<head>
    <title>YouTube Video</title>
</head>
<body>
    <!-- width/height are in % so the iframe resizes with the browser -->
    <iframe
        src="https://www.youtube.com/embed/dQw4w9WgXcQ"
        width="80%"
        height="60%"
        title="YouTube video player">
    </iframe>
</body>
</html>


-------------------- QUESTION 2 (q2.html) --------------------------
Embed a Google Map with title, width, height, loading="lazy".

<!DOCTYPE html>
<html>
<head>
    <title>My School Location</title>
</head>
<body>
    <!--
        The loading="lazy" attribute tells the browser to delay loading
        the iframe's content until it is about to scroll into view.
        This speeds up initial page load and saves bandwidth.
    -->
    <iframe
        title="Map of My School"
        src="https://www.google.com/maps/embed?pb=!1m18!1m12!1m3!1d!2d77.6!3d27.5"
        width="600"
        height="450"
        loading="lazy">
    </iframe>
</body>
</html>

</head>

        <input type="submit" value="Submit">
    </form>

        Turning off autocomplete stops the browser from storing and

        data being exposed on shared, public, or borrowed computers.
    </p>
</body>

    <!-- Internal CSS placeholder, kept empty as instructed -->

</head>
