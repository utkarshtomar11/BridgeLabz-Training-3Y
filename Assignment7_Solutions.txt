    <div class="combinator-demo">

--- FILE: q5.css ---
* { box-sizing: border-box; font-family: Arial, sans-serif; }

body { margin: 0; padding: 20px; background: #fafafa; }

header, footer { text-align: center; background: #333; color: white; padding: 12px; }

/* ATTRIBUTE SELECTORS */
input[type="text"] { border: 2px solid #3498db; }      /* matches type=text */
input[type="password"] { border: 2px solid #e74c3c; }  /* matches type=password */
a[href^="https"] { color: green; font-weight: bold; }   /* href starting with https */
a[href^="http"]:not([href^="https"]) { color: red; }    /* plain http links */

.form-demo {
  display: flex;
  gap: 15px;
  flex-wrap: wrap;
  margin: 20px 0;
}
.form-demo input { padding: 8px; border-radius: 5px; }

/* PSEUDO-CLASSES */
.item-list { list-style: none; padding: 0; margin: 20px 0; }
.item-list li {
  padding: 10px;
  background: #ecf0f1;
  margin-bottom: 5px;
  border-radius: 4px;
}
.item-list li:hover { background: #d0e6f5; }          /* :hover */
.item-list li:nth-child(odd) { border-left: 4px solid #9b59b6; } /* :nth-child */

input:focus { outline: 3px solid orange; }             /* :focus */

/* PSEUDO-ELEMENTS */
.item-list li::before { content: "➤ "; color: #3498db; } /* ::before */
.item-list li::after { content: " ✔"; color: green; }     /* ::after */

/* COMBINATORS */
.combinator-demo > p { color: #e67e22; }         /* direct child combinator */
.combinator-demo span + span { color: purple; }  /* adjacent sibling */
.combinator-demo p ~ span { text-decoration: underline; } /* general sibling */
.combinator-demo div p { font-style: italic; }   /* descendant selector */
